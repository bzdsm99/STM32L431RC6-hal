const unsigned char gImage[5000];

void lcd_show(void)
{
    uint8_t i,j;
    for(i=0;i<128;i++)
    {
        for(j=0;j<160;j++)
        {
            lcd_draw_point(i,j,LBBLUE);
            delay_ms(1);
        }
        //delay_ms(50);
    }
    delay_ms(2000);
    lcd_clear(BLACK);
    delay_ms(500);
    lcd_clear(WHITE);
    delay_ms(500);
    lcd_clear(BLUE);
    delay_ms(500);
    lcd_clear(RED);
    delay_ms(500);
    lcd_clear(YELLOW);
    delay_ms(500);
    lcd_show_pic(0,0,50,50,gImage);
    g_back_color = YELLOW;
    lcd_show_string(0,50,LCD_FONT_16,"Hello World!",RED);
    lcd_show_string(0,66,LCD_FONT_24,"STM32L4xx",BLUE);
}


int main(void)
{
    uint32_t elapsed_time, tick_time;

    HAL_Init();
    sys_stm32_clock_init(20);
    delay_init(80); //80MHZS
    // OLED_Init();
    key_init();
    led_init();
    LED_Init();
    usart_init(115200);
    // Matrix_keyboard_init();
    lcd_init();
    printf("test start...\r\n");
    tick_time = HAL_GetTick();  // 初始化时间为系统启动时间
    lcd_show();
    while (1)
    {
        elapsed_time = HAL_GetTick() - tick_time;  // 计算经过的时间
        lcd_show_num(0,spilcd_height-16,elapsed_time,8,LCD_FONT_16,NUM_SHOW_NOZERO,RED);
    }
}
